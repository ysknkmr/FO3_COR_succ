% Background for non-logicians: what FO^k and model checking are
\begin{frame}{$k$-variable logic $\mathrm{FO}^k$}

$\mathrm{FO}^k$: first-order logic with $k$ variables.

For example, in $\FOthree$:
$\varphi(x, y) := \exists z\, \bigl(E(x, y) \land E(y, z) \land E(z, x)\bigr)$ \quad \NOTE{(triangle detection)}
\begin{center}
\begin{tikzpicture}[every node/.style={font=\small}]
  % a 3-cycle through the edge x -> y, drawn as an equilateral triangle; z is existentially quantified
  \node[pv,minimum size=7pt,label=left:$x$] (x) at (210:.8) {};
  \node[pv,minimum size=7pt,label=right:$y$] (y) at (330:.8) {};
  \node[pv,minimum size=7pt,label=above:$z$] (z) at (90:.8) {};
  \draw[thick,-{Stealth},shorten >=1pt] (x) -- (y);
  \draw[thick,-{Stealth},shorten >=1pt] (y) -- (z);
  \draw[thick,-{Stealth},shorten >=1pt] (z) -- (x);
  \node[anchor=west] at (1.6,0)
    {};
\end{tikzpicture}
\end{center}

\FOthree{} also expresses, e.g.,
\begin{itemize}
  \item transitivity: $\forall x \forall y \forall z\, (E(x, y) \land E(y, z) \to E(x, z))$,
  \item closed walk of length 4: \
    $\exists x\, \exists y\, \bigl(E(x, y) \land \exists z\, (E(y, z) \land \exists y\, (E(z, y) \land E(y, x)))\bigr)$.
\end{itemize}

\vspace{1em}
\emph{Model checking}: \ $\M \models \varphi(a, b)$? \quad \NOTE{(given $\M$ with $n$ elements, a formula $\varphi$, and $a, b \in \M$)}

\pause
\vspace{.6em}
\HP{$O(|\varphi| \cdot n^{k})$ for $\mathrm{FO}^k$ by table memoization \fcite{vardi1995}}\\
E.g., $O(n^3)$ for $\varphi$ above: try all $x, y, z$
\vfill

\note{
\textbf{ページ概要}：
\begin{itemize}
  \item $\mathrm{FO}^k$ とモデル検査の問題設定
  \item 素直に計算すると $O(|\varphi| \cdot n^k)$
\end{itemize}
}
\end{frame}

\begin{frame}{Are \EMPH{$2^{O(|\varphi|)}$} compositions unavoidable?}
DNF translation is used, for example:
\begin{align*}
  & \exists z.\ \bigl(P_1(x, z) \lor Q_1(z, y)\bigr) \land \cdots \land \bigl(P_k(x, z) \lor Q_k(z, y)\bigr) \\
  \uncover<2->{\leadsto_{\text{\NOTE{(DNF)}}}} \ & \uncover<2->{\exists z.\ \bigvee_{A \subseteq [k]}
    \Bigl(\bigwedge_{i \in A} P_i(x, z) \land \bigwedge_{i \notin A} Q_i(z, y)\Bigr)} \\
  \uncover<3->{\leadsto_{\text{\NOTE{(to \CoR)}}}} \ & \uncover<3->{\Bigl(\bigcup_{A \subseteq [k]}
    \Bigl(\bigcap_{i \in A} P_i\Bigr) \mathbin{;} \Bigl(\bigcap_{i \notin A} Q_i\Bigr) \Bigr)(x, y)}
\end{align*}
\uncover<3->{\NOTE{(This translation uses $2^k$ compositions, since the number of subsets $A \subseteq [k]$ is $2^k$.)}}

\vfill
\onslide<4->{%
\begin{problem}[Open \fcite{FletcherEtAl2015,nakamura2022,nakamura-uezato2027}]
\large
Can we prove that there is no polynomial-size translation from \FOthree{} to \CoR{}?
\end{problem}
Partial answer \fcite{nakamura2022}: yes for \emph{existential} \CoR{} \fcite{nakamura2023} ($\equiv$ existential $\mathrm{FO}^3$),
\onslide<5->{via preservation under \emph{extensions}, which fails for full \CoR{}.}}

\note{
\textbf{ページ概要}：
\begin{itemize}
  \item 指数個の合成は必要不可欠か？ は未解決の問題だった
\end{itemize}
}
\end{frame}

\section{Our results}

\begin{frame}{Our results}

{\Large The answer is \emph{YES!}}
\begin{theorem}[This paper]
\label{thm:main}
\large
There is no polynomial-size translation from \FOthree{} to \CoR{}.
\pause

\normalfont
More precisely, there are \FOthree{} formulas $\varphi_1, \varphi_2, \dots$ with $|\varphi_k| = O(k)$
such that every \CoR{} term equivalent to $\varphi_k$ needs $\ge 2^{k-1}$ compositions,
even with shared subterms.
\end{theorem}

\vfill
\pause
{\large
\KEY{Also in this paper}: exponential lower bounds for
\begin{itemize}\setlength{\itemsep}{0pt}
  \item $\FOthree \leadsto \FOtwo$, binary
    \quad \NOTE{(before: $\mathrm{FO}^{k}$ vs.\ $\mathrm{FO}^{k-1}$ for $k \ge 4$ \fcite{vinall-smeeth2024})}
  \item $\CoR{} \leadsto \FOtwo$
  \pause
  \item ARA(3) $\leadsto$ matrix circuits
    \quad \NOTE{asked by \fcite{brijder2019,brijder2022}}
  \pause
  \item $\FOthree$ $\leadsto$ $\CoR{}$ with RTC \NOTE{(reflexive-transitive closure)}
\end{itemize}}

\vfill
Our proof technique applies to all of them!

\note{
\textbf{ページ概要}：
\begin{itemize}
  \item $|\varphi_k| = O(k)$ の式に、合成 $\ge 2^{k-1}$ 個が必須
  \item 同じ手法が他の設定（\FOtwo{}、行列言語、RTC 付き \CoR{} など）にも適用できる
\end{itemize}
}
\end{frame}
