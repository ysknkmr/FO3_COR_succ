\section{The question}

% ============================================================
% Paper, Sec. 1, paragraph 1
\begin{frame}{\FOthree{} is a singular point}

Three-variable first-order logic restricted to binary relations (\FOthree)
is already a singular point among the finite-variable fragments $\mathrm{FO}^k$.

\vfill
\begin{center}
\renewcommand{\arraystretch}{1.25}
\footnotesize
\begin{tabular}{@{}l>{\raggedright\arraybackslash}p{.27\textwidth}>{\raggedright\arraybackslash}p{.42\textwidth}@{}}
  \toprule
  & \FOtwo & \FOthree \\
  \midrule
  triangle, transitivity & & expresses them \\
  over words & & expresses all of first-order logic \\
  (finite) satisfiability & decidable; finite models suffice & undecidable; \EMPH{can force infinite models} \\
  property testing & & a single $\forall\exists\forall$ sentence is not testable \\
  symmetric model counting & polynomial for every sentence & $\#\mathrm{P}_1$-complete for a fixed sentence \\
  spectra & & Asser's question reduces to spectra of \FOthree{} sentences \\
  \bottomrule
\end{tabular}
\end{center}
\NOTE{\hfill satisfiability: even over a single binary relation; \ property testing: with a constant number of queries;\\
\hfill Asser's question (are first-order spectra closed under complement?) is equivalent to $\mathrm{NE} = \mathrm{coNE}$}

\vfill
\NOTE{Three variables even suffice for all of mathematics: Tarski formalized set theory in $\mathcal L^{\times}$,
a variable-free logic of equations between terms of the calculus of relations (\CoR), and these equations
have exactly the expressive power of three-variable sentences \fcite{tarski-givant1987}.}

\note{
- FO3, three-variable first-order logic restricted to binary relations, is already a singular point among the finite-variable fragments.
- Unlike FO2, it expresses the existence of a triangle and transitivity. Over words, it expresses all of first-order logic.
- Three variables even suffice for all of mathematics: Tarski formalized set theory in L-times.
- This expressive strength has sharp computational consequences: satisfiability, property testing,
  symmetric model counting, and spectra.
}
\end{frame}

\begin{frame}{Model checking: \FOthree{} is singular here as well}

Model checking is solvable in polynomial time for every $\mathrm{FO}^k$:
a $k$-variable formula $\varphi$ can be evaluated naively in time $O(|\varphi| n^k)$
\NOTE{($n$ elements)}.

\vfill
\begin{center}
\begin{tikzpicture}[every node/.style={font=\small}]
  \node[logicrow] (k4) at (0,.8) {$\mathrm{FO}^{k}$, \ $k \ge 4$};
  \node[logicrow] (k3) at (0,-.5) {\FOthree};
  \node[anchor=west,align=left] at (2,.8) {not improvable to $O(n^{k-\varepsilon})$ under the hyperclique hypothesis,\\
    \NOTE{even for a fixed sentence over a single binary relation}\\
    \NOTE{\fcite[Corollary 32]{nakamura-uezato2027}}};
  \node[anchor=west,align=left] at (2,-.5) {every sentence in time \EMPH{$2^{O(|\varphi|)}$} $\cdot\, n^{\omega}$\\
    \NOTE{$\omega < 2.3714$ \ \fcite[Theorem 10]{nakamura-uezato2027}}};
\end{tikzpicture}
\end{center}

\vfill
\pause
\FOthree{} and \CoR{} define the same binary queries \fcite{tarski-givant1987}, and the key is
the central operation of \CoR, \emph{composition}:
\begin{itemize}
  \item it eliminates the third variable, \
    $\exists z\, (P(x, z) \land Q(z, y)) \iff (P \mathbin{;} Q)(x, y)$;
  \item it is a Boolean matrix product, computable in subcubic time $O(n^{\omega})$.
\end{itemize}

\vfill
\pause
\begin{alertblock}{}
  \centering
  The factor $2^{O(|\varphi|)}$ comes from the translation into \CoR.
\end{alertblock}

\note{
- Model checking is solvable in polynomial time for every FO^k, yet FO3 is singular here as well.
- A k-variable formula can be evaluated naively in O(|phi| n^k).
- For k >= 4, under the hyperclique hypothesis, this cannot be improved to O(n^{k-eps}),
  even for a fixed sentence over a single binary relation.
- For FO3, every sentence can be evaluated in 2^{O(|phi|)} n^omega.
- FO3 and CoR define the same binary queries, and the key is composition:
  it eliminates the third variable, and it is a Boolean matrix product.
- The factor 2^{O(|phi|)} comes from the translation into CoR.
}
\end{frame}

\begin{frame}{Is the blowup necessary?}

\begin{itemize}
  \item Whether it is necessary sounds like a modern question about query compilation,
    but it is as old as Tarski's translation from the 1940s \fcite{tarski-givant1987},
    and its roots reach back to the nineteenth-century calculus of relations.
  \item It remained open
    \NOTE{(Fletcher et al.\ 2015, Remark 4.5; \cite[§8]{nakamura2022}; \cite[Question 3]{nakamura-uezato2027})},
    although under the 3-OV hypothesis no translation can be computed in polynomial time
    \fcite[Corollary 40]{nakamura-uezato2027}.
\end{itemize}

\vfill
\pause
\begin{center}
  We settle it: \ \emph{Tarski's translation is optimal in the worst case.}

  \vspace{.6em}
  Writing $T_\varphi$ for a smallest \CoR{} term equivalent to $\varphi$ on finite structures,
  \[
    \varphi \Longrightarrow T_\varphi \qquad
    |T_\varphi| = 2^{O(|\varphi|)} \ \text{(classical)}, \qquad
    \alert{|T_\varphi| = 2^{\Omega(|\varphi|)}} \ \text{(our result)}.
  \]
\end{center}

\note{
- Whether this blowup is necessary sounds like a modern question about query compilation,
  but it is as old as Tarski's translation from the 1940s.
- It remained open, although under the 3-OV hypothesis no translation can be computed in polynomial time.
- We settle it: Tarski's translation is optimal in the worst case.
}
\end{frame}

\begin{frame}{The calculus of relations (\CoR)}

On a structure with nonempty domain $\mathbb D$, \CoR{} uses the constants
\[
  \zero \defeq \emptyset, \qquad \one \defeq \mathbb D^2, \qquad \id \defeq \set{(u, u) : u \in \mathbb D},
\]
together with union, intersection, complement $\overline{X} \defeq \mathbb D^2 \setminus X$,
converse $X^{\conv} \defeq \set{(v, u) : (u, v) \in X}$, and composition
\[
  X \mathbin{;} Y \defeq \set{(u, v) : \text{some } w \in \mathbb D \text{ satisfies } (u, w) \in X,\ (w, v) \in Y}.
\]

\vfill
\begin{center}
\begin{tikzpicture}[every node/.style={font=\small}]
  \node[cellvert,minimum size=6pt,label=below:$x$] (x) at (0,0) {};
  \node[cellvert,minimum size=6pt,label=above:$z$] (z) at (1.4,.9) {};
  \node[cellvert,minimum size=6pt,label=below:$y$] (y) at (2.8,0) {};
  \draw[-{Stealth},thick] (x) -- node[above left] {$P$} (z);
  \draw[-{Stealth},thick] (z) -- node[above right] {$Q$} (y);
  \draw[-{Stealth},line width=1.5pt,black!40] (3.5,.3) -- (4.7,.3);
  \node[cellvert,minimum size=6pt,label=below:$x$] (x2) at (5.4,0) {};
  \node[cellvert,minimum size=6pt,label=below:$y$] (y2) at (8.2,0) {};
  \draw[-{Stealth},thick] (x2) -- node[above] {$P \mathbin{;} Q$} (y2);
\end{tikzpicture}
\end{center}

\vfill
Composition implements the elimination step
$\exists z\, (P(x, z) \land Q(z, y)) \iff (P \mathbin{;} Q)(x, y)$;
a vertex $z$ with $P(x, z)$ and $Q(z, y)$ is a \emph{witness} for $(x, y)$.

\note{
- To state the result formally, we fix the operations of CoR:
  the constants 0, 1, id, together with union, intersection, complement, converse, and composition.
- Composition implements the elimination step; a vertex z with P(x,z) and Q(z,y) is a witness for (x,y).
}
\end{frame}
