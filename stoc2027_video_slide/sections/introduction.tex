\begin{frame}{The calculus of relations (\CoR)}

\begin{center}
  $\zero, \ \one, \ \id, \qquad \cup, \ \cap, \ \overline{\,\cdot\,}, \ {}^{\conv}, \qquad$ and \emph{composition} \ $;$
\end{center}

\vfill
\begin{center}
\begin{tikzpicture}[every node/.style={font=\small}]
  \node[pv,minimum size=7pt,label=below:$u$] (x) at (0,0) {};
  \node[pv,minimum size=7pt,label=above:$w$] (z) at (1.8,1.1) {};
  \node[pv,minimum size=7pt,label=below:$v$] (y) at (3.6,0) {};
  \draw[thick,-{Stealth}] (x) -- node[above left] {$X$} (z);
  \draw[thick,-{Stealth}] (z) -- node[above right] {$Y$} (y);
  \draw[very thick,-{Stealth},dashed] (x) -- node[below] {$X \mathbin{;} Y$} (y);
  \node[draw=chibasub,thick,circle,minimum size=16pt,visible on=<2->] at (z) {};
  \node[anchor=west,visible on=<2->,text=chibasub] at (2.3,1.5) {a \emph{witness} for $(u, v)$};
\end{tikzpicture}
\end{center}

\vfill
\[
  X \mathbin{;} Y \defeq \set{(u, v) : \text{some } w \in \mathbb D \text{ satisfies } (u, w) \in X,\ (w, v) \in Y}
\]

\note{
\textbf{伝えること}：\CoR{} の演算を固定する。変数を消すのは合成だけで、中間点 $w$ が \emph{witness}。以降はこの言葉で話す。
\begin{itemize}
  \item 合成以外（$\cup, \cap$、補、逆、定数）は後で「自由」に使ってよいことになる
\end{itemize}
}
\end{frame}
