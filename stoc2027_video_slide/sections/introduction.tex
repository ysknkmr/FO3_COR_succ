\begin{frame}{Are \EMPH{$2^{O(|\varphi|)}$} compositions unavoidable?}
DNF translation is used, for example:
\begin{align*}
  & \exists z.\ \bigl(P_1(x, z) \lor Q_1(z, y)\bigr) \land \cdots \land \bigl(P_k(x, z) \lor Q_k(z, y)\bigr) \\
  \leadsto_{\text{\NOTE{(DNF)}}} \ & \exists z.\ \bigvee_{A \subseteq [k]}
    \Bigl(\bigwedge_{i \in A} P_i(x, z) \land \bigwedge_{i \notin A} Q_i(z, y)\Bigr) \\
  \leadsto_{\text{\NOTE{(to \CoR)}}} \ & \Bigl(\bigcup_{A \subseteq [k]}
    \Bigl(\bigcap_{i \in A} P_i\Bigr) \mathbin{;} \Bigl(\bigcap_{i \notin A} Q_i\Bigr) \Bigr)(x, y)
\end{align*}
\NOTE{(In this translation, we use $2^k$ compositions)}

\vfill
\onslide<2->{%
\begin{problem}[Open \fcite{FletcherEtAl2015,nakamura2022,nakamura-uezato2027}]
Can we prove that there is no $2^{o(|\varphi|)}$-size translation from \FOthree{} to \CoR{}?
\end{problem}
\NOTE{Partial answer: Yes for $\Sigma_1$\CoR{} (with transitive closure) \fcite{nakamura2020,nakamura2022}}}

\NOTE{The argument of \fcite{nakamura2022} is based on the fact that $\Sigma_1$\CoR{} is equivalent to 
\emph{existential} $\mathrm{FO}^3$, which is closed under \emph{extensions}.}

\note{
\textbf{スライド概要}：この指数個の合成は必要か？ 未解決だった（$\Sigma_1$\CoR{} のみ既知）。

\textbf{メモ}：
\begin{itemize}
  \item 例の式は素直に訳すと合成 $2^k$ 個
  \item 今回はサイズの無条件下界
\end{itemize}
}
\end{frame}

\begin{frame}{Our results}

{\Large The answer is \emph{YES!}}
\begin{theorem}[This paper]
\label{thm:main}
There is no $2^{o(|\varphi|)}$-size translation from \FOthree{} to \CoR{}.
\end{theorem}

\pause
\begin{theorem}[Also in this paper]
  For each $k$, there is
  \begin{enumerate}
    \item an \FOthree{} sentence of size $O(k)$ that is \FOtwo-definable,
      but every equivalent \FOtwo{} sentence has $\ge 2^{k}$ quantifiers;
      \\ \NOTE{before: $\mathrm{FO}^{k+1}$ vs.\ $\mathrm{FO}^{k}$ known for $k \ge 3$ \fcite{vinall-smeeth2024}; \ \FOthree{} vs.\ \FOtwo{} only on linear orders \fcite{grohe-schweikardt2005}}
    \item an ARA(3) expression of size $O(k)$ such that every matrix circuit computing it
      has $\ge 2^{k} / (3\beta)$ aggregation gates;
      \\ \NOTE{asked by \fcite{brijder2019,brijder2022}; \ $\beta$: join breadth of the semiring}
    \item an \FOthree{} formula of size $O(k)$ such that every equivalent \CoR{} term with closure $X^{*}$
      has $2 \cdot \#\text{compositions} + \#\text{closures} \ge 2^{k}$.
  \end{enumerate}
\end{theorem}

\vfill
\HP{Our proof technique can apply to all the above!}

\note{
\textbf{スライド概要}：答えは YES（主定理）。同じ手法で \FOtwo{}、行列言語、閉包付き \CoR{} にも指数下界。

\textbf{メモ}：
\begin{itemize}
  \item \FOtwo{}：3 変数は 2 変数より指数的に簡潔（線形順序では多項式）
  \item 行列言語：Brijder--Gyssens--Van den Bussche の問いに答える
  \item 変わるのは「1 演算が気づける変更の数」だけ（量化子 1、集約 $3\beta$、閉包 1）
  \item 論文には他に：\CoR{} 等式、逆なし回路（ちょうど $2^k$）、乱択回路、ランダム入力
\end{itemize}
}
\end{frame}
