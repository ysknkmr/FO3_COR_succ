\section{Technical overview}

% ============================================================
\begin{frame}{Proof outline (high level)}

\KEY{Goal Lemma} \ Every \CoR{} term equivalent to $G_k$ needs $2^{\Omega(k)}$ compositions.

\vfill
\KEY{Approach} We construct structures $\M_0$ and $\M_1, \dots, \M_{2^k}$ such that
\begin{itemize}
\uncover<2->{%
  \item $G_k$ distinguishes all of $\M_1, \dots, \M_{2^k}$ from $\M_0$:
    \begin{center}
    \renewcommand{\arraystretch}{1.15}
    \begin{tabular}{@{}w{l}{.6cm}|w{c}{.78cm}|*{7}{w{c}{.78cm}}|w{c}{.9cm}|w{c}{1.25cm}@{}}
      & $\M_0$ & $\M_{1}$ & $\M_{2}$ & \multicolumn{4}{w{c}{\dimexpr 3.12cm+6\tabcolsep\relax}}{$\cdots$} & $\M_{2^k}$ & \#diff. & \uncover<6->{\#comp.} \\
      \hline
      $G_k$ & T & \alert{F} & \alert{F} & \multicolumn{4}{w{c}{\dimexpr 3.12cm+6\tabcolsep\relax}}{$\cdots$} & \alert{F} & \alert{$2^k$} & \uncover<6->{$\ge 2^k/2$} \\
    \end{tabular}
    \end{center}
}
\uncover<3->{%
   \item 
\KEY{Key Lemma} \ If a \CoR{} term $t$ distinguishes $d$ structures among $\M_1, \dots, \M_{2^k}$ from $\M_0$,
then $t$ needs $\ge d/2$ compositions.
\uncover<4->{%
\begin{center}
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{@{}w{l}{.6cm}|w{c}{.78cm}|*{7}{w{c}{.78cm}}|w{c}{.9cm}|w{c}{1.25cm}@{}}
  & $\M_0$ & $\M_{1}$ & $\M_{2}$ & $\M_{3}$ & $\M_{4}$ & $\M_{5}$ & $\cdots$ & $\M_{2^k}$ & \#diff. & \#comp. \\
  \hline
  $t$ & T & \alert{F} & T & \alert{F} & T & T & $\cdots$ & T & \alert{$2$} & $\ge 1$ \\
  \uncover<5->{$t'$} & \uncover<5->{F} & \uncover<5->{\alert{T}} & \uncover<5->{\alert{T}} & \uncover<5->{\alert{T}} & \uncover<5->{\alert{T}} & \uncover<5->{F} & \uncover<5->{$\cdots$} & \uncover<5->{F} & \uncover<5->{\alert{$4$}} & \uncover<5->{$\ge 2$} \\
\end{tabular}
\end{center}
}}
\end{itemize}

\vfill
\uncover<6->{\HP{If $t \equiv G_k$, then $t$ needs $\ge 2^k/2$ compositions.}}

\vfill
\uncover<7->{\large \KEY{TASK} How can we construct such $\M_0$ and $\M_1, \dots, \M_{2^k}$?}

\note{
\textbf{ページ概要}：
\begin{itemize}
  \item 証明の方針：$G_k$ が区別する構造 $\M_0$ と $\M_1, \dots, \M_{2^k}$ を作る
  \item Key Lemma：$d$ 個を区別するには合成 $\ge d/2$ 個 $\leadsto$ $G_k$ には合成 $\ge 2^{k-1}$ 個
\end{itemize}
}
\end{frame}

% The structure M_0 (edge-colored complete graph, two middle vertices, the hub)
\begin{frame}{Structures as colored complete graphs}

\KEY{View} structures as \KEY{complete graphs}, colored with $2^k$ colors $\mu \subseteq [k]$:
$P_i$ iff $i \in \mu$.
\NOTE{$Q_i$ is used as the complement of $P_i$ (i.e., $i \notin \mu$).}

\vfill
\begin{center}
\begin{tikzpicture}[every node/.style={font=\small}]
  % same style as the figure of G_k (example with k = 3): faint candidates for z with dashed edges
  % (the dashed edges are drawn only near the candidates, to leave room for the label of x-z)
  \node[pv,minimum size=7pt,label=left:$x$] (x) at (0,0) {};
  \node[pv,minimum size=7pt,label=right:$y$] (y) at (4.8,0) {};
  \foreach \h in {2.45,3.0} {
    \node[pv,minimum size=5pt,draw=black!30,fill=black!30] (c) at (3.0,\h) {};
    \draw[densely dashed,black!30,shorten <=2.2cm] (x) -- (c);
  }
  \node[font=\footnotesize,text=black!60] at (3.45,2.72) {$\exists z$};
  \node[pv,minimum size=7pt,label=right:$z$] (z) at (3.0,1.9) {};
  \draw[thick,cA] (x) -- node[below,align=center,text=cA]
    {color $\mu = \set{1, 3}$:\\ $P_1, \ Q_2, \ P_3$} (y);
  \draw[thick,cD] (x) -- node[pos=.45,above left,align=right,text=cD,font=\footnotesize]
    {color $\mu^c = \set{2}$:\\ $Q_1, \ P_2, \ Q_3$} (z);
  \node[anchor=west,align=left] at (5.8,1.0)
    {for every $i \in [3]$,\\
    the $x$-$y$ and $x$-$z$ edges\\
    have labels $P_i$ and $Q_i$, in either order};
\end{tikzpicture}
\end{center}

\vfill
\HP{$G_k(x, y)$ holds iff some $x$-$z$ edge has the color $\mu^c$, where $\mu$ is the color of the $x$-$y$ edge.}

\note{
\textbf{ページ概要}：
\begin{itemize}
  \item 構造を色付き完全グラフとして見る
  \item $G_k(x,y)$ は $x$-$y$ の色 $\mu$ に対して、$x$-$z$ の色が $\mu^c$ であるような $z$ が存在することと同値
\end{itemize}
\textbf{口頭補足}：
\begin{itemize}
  \item 各辺で $P_i$ と $Q_i$ のちょうど一方が成り立つ
\end{itemize}
}
\end{frame}

\begin{frame}{The two-witness structure $\M_0$}

\begin{columns}[c]
  \begin{column}{.42\textwidth}
    \centering
    \input{figs/M0_random}

    \NOTE{$\M_0$: $O(k\,4^k)$ vertices, constructed by, e.g., a random coloring \fcite{jmt1995}}
  \end{column}
  \begin{column}{.55\textwidth}
    \KEY{Two-witness property} \ For all $u \ne v$ and all colors $\textcolor{cA}{\boldsymbol{\mu}}, \textcolor{cD}{\boldsymbol{\nu}}$,
    there are at least two \emph{witnesses} $\textcolor{cW}{w}$ such that
    the $u$-$w$ and $w$-$v$ edges have colors $\textcolor{cA}{\boldsymbol{\mu}}$ and $\textcolor{cD}{\boldsymbol{\nu}}$, respectively.
  \end{column}
\end{columns}

\vfill
\pause
\vspace{.3em}
\KEY{Every \CoR{} term $t$ is color-invariant on $\M_0$}: $\M_0 \models t(u, v) \iff \M_0 \models t(u', v')$
whenever the $u$-$v$ and $u'$-$v'$ edges have the same color.

\vspace{1em}
Proof by induction on $t$: the case $t = t_1 \mathbin{;} t_2$ follows from the (two-)witness property.

\vspace{.6em}
\HP{Unlike preservation under extensions (p.~3), color-invariance is preserved by complement.}

\note{
\textbf{ページ概要}：
\begin{itemize}
  \item two-witness structure $\M_0$（ランダム彩色で存在）
  \item 任意の項で同じ辺の色なら真偽が等しい（color-invariant）
\end{itemize}
}
\end{frame}

\begin{frame}{$\M_\lambda$: flipping at the hub $\hub$}

\KEY{Hub} $\hub$: a fixed arbitrary vertex of $\M_0$.

% Figure 1 of the paper (the fold), redrawn: flipping lambda turns the hub's lambda-edges into
% lambda^c-edges; afterwards the hub has no lambda-edge. Colors: lambda cL, lambda^c cA.
\begin{center}

  \foreach \cx/\name/\lab/\vis in {0/0/structure $\M_0$/1-,6.64/1/flipped structure $\M_\lambda$/2-} {
    \begin{scope}[shift={(\cx,1.2)},visible on=<\vis>]

      \foreach \t in {0,30,60,90,120,150,180,210,240,270,300,330} {

      }

      \node[pv] (o\name) at (90:1.2) {};
      \draw[thick,black!60] (h\name) -- node[lbl,pos=.4,text=black!60] {$\mu$ ($\ne \lambda$)} (o\name);
      \draw[thick,cA] (h\name) -- node[lbl,pos=.55,text=cA] {$\lambda^c$} (v\name);

  }

  \draw[thick,cL] (h0) -- node[lbl,pos=.6,text=cL] {$\lambda$} (a0);
  \draw[thick,cL] (h0) -- node[lbl,pos=.6,text=cL] {$\lambda$} (w0);
  \node[font=\normalsize] at (0,-.85) {$\hub$ has $\lambda$-edges};
  % M_lambda: the same edges are still there, now of color lambda^c (as the h-v edge)
  \begin{scope}[visible on=<2->]
    \draw[thick,cA] (h1) -- node[lbl,pos=.6,text=cA] {$\lambda^c$} (a1);

    \node[font=\normalsize,align=center] at (6.64,-1.0) {$\hub$ \alert{does not} have $\lambda$-edges};
    \draw[-{Stealth},thick] (2.6,1.2) -- node[above] {flip $\lambda$} (4.0,1.2);
  \end{scope}
\end{tikzpicture}
\end{center}

\vfill
\onslide<3->{%
% $\M_\lambda$ differs from $\M_0$ only in the $\lambda$-edges at the hub:\\
For $v$ with the $\hub$-$v$ edge of color $\lambda^c$: \ $\M_0 \models G_k(\hub, v)$ \ but \ $\M_\lambda \not\models G_k(\hub, v)$.}

\vfill
\onslide<4->{%
Flipping each of the $2^k$ colors $\lambda$ gives $\M_1, \dots, \M_{2^k}$.
% Each $\M_\lambda$ differs from $\M_0$ only at the hub, in one color.
}

\onslide<5->{%
\KEY{Definition} \ $r$ \emph{distinguishes} $\M_\lambda$ from $\M_0$ if
$\M_0 \models r(u, v)$ but $\M_\lambda \not\models r(u', v')$ (or vice versa)
for some $u$-$v$ edge in $\M_0$ and $u'$-$v'$ edge in $\M_\lambda$ of the same color.}

\onslide<6->{%
$\leadsto$ Each composition newly distinguishes \alert{at most two} of them \NOTE{(next slides)}.
}
\note{
\textbf{ページ概要}：
\begin{itemize}
  \item hub の $\lambda$-edges を flip した $\M_\lambda$ を考えると $2^k$ 個の構造を得る。
  \item 各合成が新たに区別するのは高々 2 個（次の 2 ページ）
\end{itemize}
\textbf{口頭補足}：
\begin{itemize}
  \item 補・逆・和・積は点対ごとの演算なので、新たに区別することはない（口頭）
\end{itemize}
}
\end{frame}

\begin{frame}{Why a composition newly distinguishes at most two $\M_\lambda$ \ (1/2)}

\KEY{Q.} \ If $t$ and $s$ do not distinguish $\M_\lambda$ from $\M_0$, when can $t \mathbin{;} s$ newly distinguish $\M_{\lambda}$?

\medskip
\KEY{Case $\hub \notin \{u, v\}$:} \ Never: $t \mathbin{;} s$ cannot distinguish $\M_{\lambda}$ from $\M_0$ at such $(u, v)$.

% M_0 and M_lambda differ only in the edges at the hub (dashed on both sides);
% for every color pair, a witness can be re-picked away from the hub.
\begin{center}
% trim: center the two panels (the annotations may stick out)
\begin{tikzpicture}[every node/.style={font=\small},x=1.3cm,y=.85cm,
    trim left={(-.35,0)}, trim right={(6.4/1.3+4.3,0)},
    lbl/.style={fill=normal text.bg,inner sep=2pt,font=\footnotesize},
    hubwit/.style={wit={$\hub$}}, hubplain/.style={nowit={$\hub$}},
    zwit/.style={witon=<4->{$w$}}, zplain/.style={nowit={$w$}}]
  \begin{scope}[visible on=<2->]
    % M_0: the witness is the hub; M_lambda: re-picked away from the hub, same color pair
    \foreach \dx/\str/\hst/\zst in {0/{\M_0}/hubwit/zplain, 6.4/{\M_\lambda}/hubplain/zwit} {
      \begin{scope}[xshift=\dx cm]
        \node at (1,1.75) {$\str$};
        \node[pairv] (u) at (0,0) {$u$};
        \node[pairv] (v) at (2,0) {$v$};
        \node (h) at (1,1.0) [\hst];
        \begin{scope}[visible on=<3->]
          \node (w) at (1,-1.0) [\zst];
          \draw[thick] (u) -- node[lbl] {$\mu$} (w);
          \draw[thick] (w) -- node[lbl] {$\nu$} (v);
        \end{scope}
      \end{scope}
    }
    \begin{pgfonlayer}{background}
      % (a layer does not inherit the scope's overlay, so repeat visible on)
      \begin{scope}[visible on=<2->]
      \draw[thick,black!35,densely dashed] (0,0) -- node[lbl,text=black] {$\mu$} (1,1.0);
      \draw[thick,black!35,densely dashed] (1,1.0) -- node[lbl,pos=.78,text=black] {$\nu$} (2,0);
      \draw[thick,black!35,densely dashed] (6.4/1.3,0) -- (6.4/1.3+1,1.0) -- (6.4/1.3+2,0);

    \draw[-{Stealth},thick,black!40] (2.65,0) -- node[above,font=\scriptsize,text=black!50] {flip $\lambda$} (4.0,0);

    \node[stepno,anchor=north east] at ([xshift=-2pt,yshift=-3pt]n1.north west) {1};
    \node[anchor=west,align=left,font=\footnotesize,text=cW,visible on=<4->] (n3) at (6.4/1.3+1.52,-1.2)
      {We can pick $w$ as\\ the corresponding witness};
    \node[stepno,anchor=north east,draw=cW,text=cW,visible on=<4->] at ([xshift=-2pt,yshift=-3pt]n3.north west) {3};
    % \node[anchor=west,align=left,font=\footnotesize,text=black!60] at (6.3,1.1) {dashed edges at $\hub$\\ are possibly changed by the flip};
  \end{scope}
  \node[font=\footnotesize,text=black!60,visible on=<3->,align=left,anchor=north west] (n2) at (-.1,-1.45)
    {
    The \KEY{two-witness property} shows\\

  \node[stepno,anchor=north east,visible on=<3->] at ([xshift=-2pt,yshift=-3pt]n2.north west) {2};

\onslide<5->{\NOTE{Similarly for the converse direction (from $\M_\lambda$ to $\M_0$).}}

\note{
\textbf{ページ概要}：
\begin{itemize}
  \item hub を含まない点対では、同じ色の組の witness を別の頂点で取れる（two-witness property）
  \item よってこの場合では区別できない
\end{itemize}
}
\end{frame}

\begin{frame}{Why a composition newly distinguishes at most two $\M_\lambda$ \ (2/2)}

\KEY{Case $u = \hub$} (similarly for $v = \hub$): \ the $\hub$-$w$ edge itself may be flipped.

% Same layout as (1/2): (1) a witness w via a lambda-edge at the hub in M_0; the flip turns it into
% lambda^c; (2) if t also holds on mu != lambda, there is w' with colors (mu, nu) (two-witness
% property), and its mu-edge at the hub is not flipped; (3) w' is the corresponding witness.
% Colors as on the flipping slide: lambda cL, lambda^c cA.
\begin{center}
% trim: center the two panels (the annotations may stick out)
\begin{tikzpicture}[every node/.style={font=\small},x=1.3cm,y=.95cm,
    trim left={(-.35,0)}, trim right={(6.4/1.3+4.3,0)},
    lbl/.style={fill=normal text.bg,inner sep=2pt,font=\footnotesize},
    wA/.style={wit={$w$}}, wplain/.style={nowit={$w$}},
    wpA/.style={nowit={$w'$}}, wpB/.style={witon=<4->{$w'$}}]
  \begin{scope}[visible on=<2->]
    \foreach \dx/\str/\col/\clab/\wst/\wpst in {0/{\M_0}/cL/{\lambda}/wA/wpA, 6.4/{\M_\lambda}/cA/{\lambda^c}/wplain/wpB} {
      \begin{scope}[xshift=\dx cm]
        \node at (1,1.75) {$\str$};
        \node[pairv] (h) at (0,0) {$\hub$};
        \node[pairv] (v) at (2,0) {$v$};
        \node (w) at (1,1.0) [\wst];
        \draw[thick,\col] (h) -- node[lbl,text=\col] {$\clab$} (w);
        \draw[thick] (w) -- node[lbl,pos=.55] {$\nu$} (v);

        }
      \end{scope}
    }
    \draw[-{Stealth},thick,black!40] (2.65,0) -- node[above,font=\scriptsize,text=black!50] {flip $\lambda$} (4.0,0);
    \node[anchor=west,font=\footnotesize] (n1) at (1.55,1.0) {$w$ picked as a witness};
    \node[stepno,anchor=east] at ([xshift=-2pt]n1.west) {1};
  \end{scope}
  \node[anchor=north west,align=left,font=\footnotesize,visible on=<3->] (n2) at (6.4/1.3+.15,-1.3)
    {Here we cannot pick a witness\\
    with the same colors $(\lambda, \nu)$};
  \node[stepno,anchor=north east,visible on=<3->] at ([xshift=-2pt,yshift=-3pt]n2.north west) {2};
\end{tikzpicture}
\end{center}

\onslide<4->{%
So the answer changes only if $t$ holds on some $\lambda$-edges and only on them \NOTE{($v = \hub$: the same for $s$)}.
}
\onslide<5->{%
Such $\lambda$ is uniquely determined for $t$ and for $s$.}

\onslide<6->{
$\leadsto$ Each composition newly distinguishes \alert{at most two} $\M_\lambda$.\\

\NOTE{(The remaining cases: loops $(u, u)$, including $(\hub, \hub)$, and witnesses $w \in \{u, v\}$. See the paper for details.)}}

\note{
\textbf{ページ概要}：
\begin{itemize}
  \item hub を含む点対で区別できるのは、$t$（または $s$）が $\lambda$-edges だけで成り立つときだけ
  \item 各合成が新たに区別できる $\M_\lambda$ は高々 2 個
\end{itemize}
}
\end{frame}
