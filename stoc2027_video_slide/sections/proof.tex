\begin{frame}{Proof idea: telling $\M_0$ apart from $2^k$ structures}

\KEY{Goal} \ $C \equiv G_k$ \ $\Longrightarrow$ \ $C$ has $\ge 2^{k-1}$ compositions
\NOTE{\ ($C$: any \CoR{} term or circuit)}

\vspace{.3em}
\uncover<2->{%
$C$ must tell $\M_0$ apart from each of $2^k$ structures $\M_\lambda$:
\ $G_k = 1$ on $\M_0$, \ $G_k = 0$ on $\M_\lambda$}

\begin{center}
\begin{tikzpicture}[every node/.style={font=\small}]
  \node (M0) at (6.1,1.0) {$\M_0$};
  \foreach \i in {1,...,8} {
    \node[visible on=<2->] (L\i) at (1.4+\i*1.05,0) {$\M_{\lambda_{\i}}$};
    \draw[densely dashed,black!40,visible on=<2->] (M0.south) -- (L\i.north);
  }
  \node[visible on=<2->] at (11.0,0) {$\cdots$};
  % each composition newly tells M_0 apart from at most two of them
  \foreach \i/\j/\g in {1/2/1,3/4/2,5/6/3,7/8/4} {
    \draw[decorate,decoration={brace,mirror,amplitude=4pt},thick,chibasub,visible on=<4->]
      ($(L\i.south west)+(.05,-.05)$) -- ($(L\j.south east)+(-.05,-.05)$)
      node[midway,below=4pt,font=\scriptsize] {composition $;_{\g}$};
  }
\end{tikzpicture}
\end{center}

\uncover<3->{%
\begin{center}
\small
\renewcommand{\arraystretch}{1.05}
\begin{tabular}{@{}lll@{}}
  \toprule
  \CoR{} operation & meaning & newly tells $\M_0$ apart from \\
  \midrule
  $\zero, \one, \id$, \ $\cup, \cap, \overline{\,\cdot\,}$, \ $X^{\conv}$ & constants, Boolean operations, converse & no $\M_\lambda$ \\
  $X \mathbin{;} Y$ & composition $=$ Boolean matrix product & \alert{$\le 2$ of the $\M_\lambda$} \\
  \bottomrule
\end{tabular}
\end{center}}

\uncover<4->{%
\begin{center}
  every $\M_\lambda$ needs a composition that tells it apart
  \quad $\Longrightarrow$ \quad \alert{$2^k \le 2 \cdot \#\text{compositions}(C)$}
\end{center}}

\note{
\textbf{スライド概要}：$C$ は $\M_0$ と $2^k$ 個の $\M_\lambda$ を見分けねばならない。合成 1 個が新たに見分けられるのは高々 2 個。

\textbf{メモ}：
\begin{itemize}
  \item 「新たに見分ける」＝そのゲートで初めて $\M_0$ から期待される値とずれる（論文の notice）
  \item 以降のページ：変更（塗り替え）の作り方、期待値の意味、合成が高々 2 つの理由
\end{itemize}
}
\end{frame}

\begin{center}
  $G_k(\hub, v)$ needs an edge $\hub z$ of color $\lambda$
  \onslide<2->{\\ recoloring $\lambda$ changes \emph{exactly those edges at $\hub$}, \ and nothing else}
\end{center}

\onslide<3->{%
\begin{center}
  $2^k$ recolored structures $\M_\lambda$, one for each color $\lambda$
\end{center}}

  \item 色ごとに 1 つ、計 $2^k$ 個

\begin{frame}{Why a composition detects at most two recolorings}

% Figure 2 of the paper, redrawn with one layout for all panels:
% the pair is the bottom edge, witnesses sit above (and below) it.
% Red with a cross: an edge recolored by the fold; thick green: a witness that survives.
\begin{center}
\begin{tikzpicture}[every node/.style={font=\small},x=1.2cm,y=1.2cm,
    xmark/.style={text=red,font=\large\bfseries,inner sep=0pt},
    lost/.style={thick,red,densely dashed},
    kept/.style={line width=1.4pt,cC}]
  % (a) M_0
  \begin{scope}
    \node at (.9,1.75) {(a) $\M_0$, at $(\hub, v)$};
    \node[pv,label=below:$\hub$] (h) at (0,0) {};
    \node[pv,label=above:$z$] (z) at (.9,1.0) {};
    \node[pv,label=below:$v$] (v) at (1.8,0) {};
    \draw[thick] (h) -- node[above left] {$\lambda$} (z);
    \draw[thick] (z) -- node[above right] {$\nu$} (v);
    \draw[densely dotted] (h) -- (v);
    \node[align=center] at (.9,-.85) {middle vertex $z$:\\ $(X \mathbin{;} Y)(\hub, v) = 1$};
  \end{scope}
  % (b) M_lambda, at the hub: every witness uses an edge at the hub
  \begin{scope}[xshift=4.4cm,visible on=<2->]
    \node at (.9,1.75) {(b) $\M_\lambda$, at $(\hub, v)$};
    \node[pv,label=below:$\hub$] (h) at (0,0) {};
    \node[pv,label=above:$z$] (z) at (.9,1.0) {};
    \node[pv,label=below:$v$] (v) at (1.8,0) {};
    \draw[lost] (h) -- node[above left] {$\lambda^c$} (z);
    \node[xmark] at ($(h)!.5!(z)$) {$\times$};
    \draw[thick] (z) -- node[above right] {$\nu$} (v);
    \draw[densely dotted] (h) -- (v);
    \node[align=center] at (.9,-.85) {every middle vertex is lost:\\ \alert{detected}};
  \end{scope}
  % (c) M_lambda, away from the hub: the hub is only one of the witnesses
  \begin{scope}[xshift=8.8cm,visible on=<3->]
    \node at (.9,1.75) {(c) $\M_\lambda$, at $(u, v)$};
    \node[pv,label=left:$u$] (u) at (0,0) {};
    \node[pv,label=above:$\hub$] (h) at (.9,1.0) {};
    \node[pv,label=right:$v$] (v) at (1.8,0) {};
    \node[pv,label=below:$z'$] (z) at (.9,-.8) {};
    \draw[lost] (u) -- node[above left] {$\lambda^c$} (h);
    \node[xmark] at ($(u)!.5!(h)$) {$\times$};
    \draw[thick] (h) -- node[above right] {$\nu$} (v);
    \draw[kept] (u) -- node[below left] {$\lambda$} (z);
    \draw[kept] (z) -- node[below right] {$\nu$} (v);
    \draw[densely dotted] (u) -- (v);
    \node[align=center] at (.9,-1.45) {a second middle vertex $z'$\\ avoids $\hub$: \ not detected};
  \end{scope}
\end{tikzpicture}
\end{center}
\NOTE{\hfill a composition $X \mathbin{;} Y$, \ where $X$ holds on the edges of color $\lambda$ and $Y$ on those of color $\nu \ne \lambda$}

\vfill
\onslide<4->{%
\begin{center}
  only pairs at the hub can change, and only if $X$ or $Y$ singles out the color $\lambda$
  \\ $\Rightarrow$ \ each composition detects at most \alert{two} colors $\lambda$
\end{center}}

\note{
\textbf{スライド概要}：$\hub$ を含む点対では気づかれうる、含まない点対では気づかれない。

\textbf{メモ}：
\begin{itemize}
  \item $\hub$ を含まない点対では、$\hub$ を通らない中間の頂点 $z'$ が残る
  \item 条件は 3 つ、どれも $\lambda$ を決める $\to$ 高々 2
\end{itemize}
}
\end{frame}
