\section{Technical overview}

% ============================================================
\begin{frame}{Our technique: exact predictions on one reference structure}

\begin{enumerate}
  \item Modify \emph{one finite structure} $\M_0$ in \emph{exponentially many small ways}
    $\M_\lambda$ \ ($\lambda \subseteq [k]$), each changing the answer of $G_k$.
  \vfill
  \pause
  \item Compare a circuit on $\M_\lambda$ with what its values on $\M_0$ \emph{predict},
    by an \emph{equality}: only compositions can notice a modification.
  \vfill
  \pause
  \item Each composition can notice at most \emph{two} modifications.
\end{enumerate}

\vfill
\pause
\[
  2^k \;\le\; 2 \cdot \gcost{C}.
\]

\note{
- Our technique in one sentence: exact predictions on one reference structure.
- We modify one finite structure in 2^k small ways, each changing the answer of G_k.
- We compare the circuit on each modification with the values predicted from the reference structure;
  this is an equality, which survives complement, and only compositions can notice.
- Each composition notices at most two modifications, which gives the bound.
}
\end{frame}

% ============================================================
\begin{frame}{The reference structure}

Edge-colored complete graphs $\M = (V, \colr)$, \ $\colr\colon \binom{V}{2} \to 2^{[k]}$,
\ $\colr(u, u) \defeq \id$ \NOTE{(a label, not a color)}.
\[
  \text{on an edge of color } \mu: \quad P_i \iff i \in \mu, \qquad Q_i \iff i \notin \mu
  \qquad \NOTE{(no input on loops)}
\]
\HP{a read edge of color $\mu$ accepts exactly the witness edges of color $\mu^c$.}

\vfill
\pause
\begin{columns}[c]
\begin{column}{.26\textwidth}
\centering
\begin{tikzpicture}[every node/.style={font=\small},scale=.7]
  \node[cellvert,label=left:$u$] (u) at (0,0) {};
  \node[cellvert,label=right:$v$] (v) at (3.2,0) {};
  \node[cellvert,label=above:$z$] (z1) at (1.6,1.3) {};
  \node[cellvert,label=below:$z'$] (z2) at (1.6,-1.3) {};
  \draw[thick] (u) -- node[above left] {$\mu$} (z1);
  \draw[thick] (z1) -- node[above right] {$\nu$} (v);
  \draw[thick] (u) -- node[below left] {$\mu$} (z2);
  \draw[thick] (z2) -- node[below right] {$\nu$} (v);
\end{tikzpicture}
\end{column}
\begin{column}{.72\textwidth}
\emph{Two-witness property}: for distinct $u, v$ and all $\mu, \nu$, at least two
$z \notin \set{u, v}$ have $\colr(u, z) = \mu$, $\colr(z, v) = \nu$.\\
\NOTE{Exists for $|V| \ge n_0(2^k)$: a random coloring \fcite{jmt1995}.}

\vspace{.3em}
Fix one: the \emph{reference structure} $\M_0$, and a vertex $\hub$, the \emph{hub}.
\end{column}
\end{columns}

\vfill
\pause
\begin{itemize}
  \item[(M1)] every vertex has edges of every color;
  \item[(M2)] \alert{the value at $(u, v)$ of every relation that a circuit computes on $\M_0$
    depends only on $\colr(u, v)$};
  \item[(M3)] for distinct $u, v \ne \hub$ and all $\mu, \nu$, some witness $z \notin \set{u, v, \hub}$ exists.
\end{itemize}

\note{
- Our structures are edge-colored complete graphs; the colors are the subsets of [k],
  and the color of an edge determines the inputs on it.
- So a read edge of color mu accepts exactly the witness edges of the complementary color.
- The two-witness property: every pair of colors is realized twice between any two vertices.
  A random coloring has it once the graph is large enough.
- Three consequences recur: every vertex sees every color; values computed on M_0 depend only on colors;
  and witnesses can avoid the hub.
}
\end{frame}

\begin{frame}{Folds}

The \emph{fold of $\lambda$} recolors the edges of color $\lambda$ at $\hub$ to $\lambda^c$;
the result is the \emph{folded structure} $\M_\lambda = (V, \colr_\lambda)$.

\vfill
\pause
By (M1) fix $v_{\lambda^c}$ with $\colr(\hub, v_{\lambda^c}) = \lambda^c$; the fold does not recolor this read edge.
It accepts exactly witness edges of color $\lambda$, which $\hub$ has in $\M_0$ but not in $\M_\lambda$.

\note{
- For each color lambda, the fold of lambda recolors the lambda-edges at the hub to the complementary color.
- The read edge from the hub to v_{lambda^c} keeps its color, and it accepts exactly witness edges of color lambda.
- The hub has such edges in M_0, but not in M_lambda, where the fold has recolored all of them.
- We get 2^k folds, one for each color.
}
\end{frame}

\begin{frame}{Predictions and noticing}

Compare a circuit $C$ on $\M_0$ and $\M_\lambda$ gate by gate:
\vspace{-.3em}
\[
  \sem{g}{\M_\lambda}(u, v) = \sem{g}{\M_0}(u', v') \quad \text{whenever } \colr_\lambda(u, v) = \colr(u', v').
  \tag{$\star$}
\]
\NOTE{By (M2) the right side depends only on the color: what $\M_0$ \emph{predicts} for the new colors.}

\begin{itemize}
  \item every leaf satisfies $(\star)$; \ an output computing $G_k$ violates it at $(\hub, v_{\lambda^c})$.
\end{itemize}

\vfill
\pause
A gate $g$ \emph{notices} the fold of $\lambda$ if $g$, applied on $\M_\lambda$ to the values predicted
for its operands, does not compute the value predicted for $g$.
\NOTE{This depends only on the operation of $g$ and the values of its operands on $\M_0$.}
\[
  \Delta_g \defeq \set{\lambda : g \text{ violates } (\star)}, \quad
  D_g \defeq \set{\lambda : g \text{ notices the fold of } \lambda}, \quad
  \Delta_g \subseteq \Delta_{g_1} \cup \Delta_{g_2} \cup D_g.
\]
\[
  \emptyset = \Delta_{\mathrm{leaf}} \rightsquigarrow \cdots \rightsquigarrow \Delta_{\mathrm{output}} = 2^{[k]}
  \quad\Rightarrow\quad \text{every color lies in some } D_g.
\]

\vfill
\pause
\begin{itemize}
  \item Unlike $(\diamond)$, the equality $(\star)$ \emph{survives complement}: Boolean operations and converse notice nothing.
  \item Each composition notices at most two folds \NOTE{(next)} \ $\Rightarrow$ \ $\gcost{C} \ge 2^k / 2$;
    \ $D_g$ is attached to the gate, so this holds \emph{with sharing}.
\end{itemize}

\note{
- We compare the two computations by an equality: on M_lambda, each gate should compute what its values on M_0 predict from the new colors.
- Leaves satisfy it, because inputs are determined by colors in the same way. The output for G_k violates it.
- A violation at g is inherited from an operand or noticed by g itself,
  so every color lies in D_g for some gate g.
- D_g depends only on values on M_0, not on the computation on the folded structures.
- The equality survives complement: Boolean operations and converse notice nothing. Only compositions notice folds, at most two each.
}
\end{frame}

\begin{frame}{Values on the reference structure are color tables}

$X$ has a \emph{color table}: $X(u, v)$ depends only on $\colr(u, v)$;
\ $X[\mu]$, $X[\id]$; \ $S_X \defeq \set{\mu : X[\mu] = 1}$.

\vfill
\begin{columns}[c]
\begin{column}{.36\textwidth}
\centering
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{@{}lccc@{}}
  \toprule
  $X$ & $X[\id]$ & $X[\set{1}]$ & $X[\emptyset]$ \\
  \midrule
  $P_1$ & 0 & 1 & 0 \\
  $Q_1$ & 0 & 0 & 1 \\
  $P_1 \mathbin{;} Q_1$ & 0 & 1 & 1 \\
  $P_1 \mathbin{;} \one$ & 1 & 1 & 1 \\
  \bottomrule
\end{tabular}

\NOTE{$k = 1$}
\end{column}
\begin{column}{.62\textwidth}
\pause
Boolean operations and converse act per color. For composition, by two witnesses:
\begin{align*}
  (X \mathbin{;} Y)[\mu] &= (X[\id] \land Y[\mu]) \lor (X[\mu] \land Y[\id])\\
                         &\quad \lor (S_X \ne \emptyset \land S_Y \ne \emptyset),\\
  (X \mathbin{;} Y)[\id] &= (X[\id] \land Y[\id]) \lor (S_X \cap S_Y \ne \emptyset).
\end{align*}
\end{column}
\end{columns}

\vfill
\pause
\begin{alertblock}{Folded structures need not have color tables}
  $\sem{H_\lambda \mathbin{;} \one}{\M_\lambda}$ fails at every $(\hub, v)$, but holds at the other pairs of the same colors.
  \ $\Rightarrow$ \ compare each $\M_\lambda$ with $\M_0$ \emph{directly}.
\end{alertblock}

\note{
- (M2): every relation computed on M_0 has a color table, however much complement is used.
- Composition: endpoint witnesses depend on the entries at the label; an outside witness exists
  iff both operands hold on some color, or, on loops, on a common color.
- On folded structures, tables fail: this is why (star) refers to M_0,
  and we never compare two folded structures.
}
\end{frame}

% ============================================================
\begin{frame}{Predictions}

For $X$ with a color table, its \emph{prediction} on $\M_\lambda$ is
\[
  \pred{X}{\lambda}(u, v) \defeq X[\colr_\lambda(u, v)],
  \qquad \text{so } (\star) \iff \sem{g}{\M_\lambda} = \pred{(\sem{g}{\M_0})}{\lambda}.
\]

\vfill
\pause
A composition with operand values $X, Y$ notices the fold of $\lambda$ iff
\[
  \pred{(X \mathbin{;} Y)}{\lambda} \;\ne\; \pred{X}{\lambda} \mathbin{;} \pred{Y}{\lambda}.
\]
Others notice nothing:
\ $\overline{\pred{X}{\lambda}} = \pred{\overline{X}}{\lambda}$, \
$\pred{X}{\lambda} \cup \pred{Y}{\lambda} = \pred{(X \cup Y)}{\lambda}$, \
$(\pred{X}{\lambda})^{\conv} = \pred{(X^{\conv})}{\lambda}$, \
$\zero, \one, \id$ are their own predictions.

\vfill
\pause
\NOTE{Two simpler invariants fail:
$\sem{g}{\M_\lambda} = \sem{g}{\M_0}$ already fails at the leaves;
equality outside the recolored edges is not inductive, since a composition moves values from those edges to other pairs.}

\note{
- The prediction of X reads its table with the new colors.
- A composition notices a fold exactly when composing the predictions differs from predicting the composition.
- Boolean operations, converse, and constants commute with predictions.
- Two simpler invariants would fail: the equality of the two values already fails at the leaves,
  and equality outside the recolored edges cannot be carried through an induction.
}
\end{frame}

% ============================================================
\begin{frame}{Each composition notices at most two folds}

\begin{lemma}[Each composition notices at most two folds]
  For any two relations $X, Y$ on the reference structure that have color tables,
  $\pred{(X \mathbin{;} Y)}{\lambda} = \pred{X}{\lambda} \mathbin{;} \pred{Y}{\lambda}$
  for all colors $\lambda$ except at most two.
\end{lemma}
\NOTE{A single pair of color tables; no circuit is involved.}

\vfill
\pause
\begin{alertblock}{On an arbitrary structure}
  $R = \set{(a, b_i)}_{i \le t}$, \ $S = \set{(b_i, c_i)}_{i \le t}$:
  removing $(a, b_i)$ from $R$ removes $(a, c_i)$ from $R \mathbin{;} S$.\\
  A single composition reacts to $t$ separate changes
  \NOTE{(much as a single term separated all of Nakamura's instances)}.
\end{alertblock}

\note{
- Lemma 1.4 concerns a single pair of color tables; no circuit is involved.
- On arbitrary structures a single composition can react to many separate changes;
  on the reference structure it cannot.
}
\end{frame}

\begin{frame}{Proof of Lemma 1.4}

Endpoint witnesses contribute the same to both sides; only outside witnesses matter.
The two sides can differ only
\begin{center}
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{@{}lll@{}}
  \toprule
  pair & why & condition \\
  \midrule
  $(\hub, v)$ & edges $\set{\hub, z}$ carry every color except $\lambda$ & $S_X = \set{\lambda}$ \\
  $(u, \hub)$ & symmetric & $S_Y = \set{\lambda}$ \\
  $(\hub, \hub)$ & a witness uses the single edge $\set{\hub, z}$ twice & $S_X \cap S_Y = \set{\lambda}$ \\
  $u, v \ne \hub$ & (M3): a witness $z' \notin \set{u, v, \hub}$ & never \\
  \bottomrule
\end{tabular}
\end{center}
Each condition determines $\lambda$; if $S_X \cap S_Y \ne \emptyset$, all force $S_X \cap S_Y = \set{\lambda}$.
\ $\Rightarrow$ \ at most \emph{two} folds.
\NOTE{(At most one for a row projection $X \mathbin{;} \one$.)}

\note{
- At the hub, after the fold, the edges at the hub carry every color except lambda.
  So the sides differ only if lambda is the only color of X, or of Y, or of their intersection.
- Away from the hub, (M3) still provides a witness z' outside {u, v, h}, so the two sides agree.
- Each condition determines lambda, so at most two folds are noticed.
- This finishes the proof of Theorem 1.1.
}
\end{frame}
