\begin{frame}{The two-witness property}

\begin{center}
\begin{tikzpicture}[every node/.style={font=\small}]
  \node[pv,label=left:$u$] (u) at (0,0) {};
  \node[pv,label=right:$v$] (v) at (4,0) {};
  \node[pv,label=above:$z$] (z1) at (2,1.3) {};
  \node[pv,label=below:$z'$] (z2) at (2,-1.3) {};
  \draw[thick,cA] (u) -- node[above left] {$\mu$} (z1);
  \draw[thick,cB] (z1) -- node[above right] {$\nu$} (v);
  \draw[thick,cA] (u) -- node[below left] {$\mu$} (z2);
  \draw[thick,cB] (z2) -- node[below right] {$\nu$} (v);
\end{tikzpicture}
\end{center}
\begin{center}
  for all $u \ne v$ and all colors $\mu, \nu$: \ at least two witnesses
\end{center}

\vfill
\pause
\begin{center}
  exists for $n \ge n_0(2^k)$ \NOTE{(random coloring; \cite{jmt1995})} \\[.4em]
  fix one: the \emph{reference structure} $\M_0$, \ and a vertex $\hub$: the \emph{hub}
\end{center}

\note{
\textbf{伝えること}：参照構造の唯一の要請は「どの点対・どの色の組にも witness が 2 つ以上ある」こと。
\begin{itemize}
  \item ランダム彩色で $n \ge n_0(2^k)$ なら存在する（Jipsen--Maddux--Tuza）
  \item 1 つ固定して $\M_0$、さらに任意の 1 頂点を hub $\hub$ とする
  \item 「2 つ」が効くのは後半：hub を避けた witness が必ず残る
\end{itemize}
}
\end{frame}

\end{itemize}

\begin{frame}{Predictions}

  \pred{X}{\lambda}(u, v) \defeq X[\colr_\lambda(u, v)]
  \qquad \text{\NOTE{(read the table of $X$ with the new colors)}}
\]

\vspace{1.5em}

\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{@{}l@{\qquad}l@{}}
  $\overline{\pred{X}{\lambda}} = \pred{\overline{X}}{\lambda}$, \ $\pred{X}{\lambda} \cup \pred{Y}{\lambda} = \pred{(X \cup Y)}{\lambda}$, \ \dots
    & notice nothing \\
  \onslide<3->{$\pred{X}{\lambda} \mathbin{;} \pred{Y}{\lambda} \;\overset{?}{=}\; \pred{(X \mathbin{;} Y)}{\lambda}$}

\end{itemize}

\begin{frame}{Each composition notices at most two folds}

\begin{lemma}[Each composition notices at most two folds]
  For color tables $X, Y$: \
  $\pred{(X \mathbin{;} Y)}{\lambda} = \pred{X}{\lambda} \mathbin{;} \pred{Y}{\lambda}$
  for all $\lambda$ except at most two.
\end{lemma}

\vfill

\note{

\end{itemize}
