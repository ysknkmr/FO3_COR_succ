\section{Technical overview}

% ============================================================
\begin{frame}{Proof outline (at high-level)}

\KEY{Goal Lemma} \ Every \CoR{} term equivalent to $G_k$ needs $2^{\Omega(k)}$ compositions.

\vfill
We construct structures $\M_0$ and $\M_1, \dots, \M_{2^k}$ such that
\begin{itemize}
\uncover<2->{%
  \item $G_k$ distinguishes all of $\M_1, \dots, \M_{2^k}$ from $\M_0$:
    \begin{center}
    \renewcommand{\arraystretch}{1.15}
    \begin{tabular}{@{}w{l}{.6cm}|w{c}{.78cm}|*{7}{w{c}{.78cm}}|w{c}{.9cm}|w{c}{1.25cm}@{}}
      & $\M_0$ & $\M_{1}$ & $\M_{2}$ & \multicolumn{4}{w{c}{\dimexpr 3.12cm+6\tabcolsep\relax}}{$\cdots$} & $\M_{2^k}$ & \#diff. & \uncover<6->{\#comp.} \\
      \hline
      $G_k$ & T & \alert{F} & \alert{F} & \multicolumn{4}{w{c}{\dimexpr 3.12cm+6\tabcolsep\relax}}{$\cdots$} & \alert{F} & \alert{$2^k$} & \uncover<6->{$\ge 2^k/2$} \\
    \end{tabular}
    \end{center}
}
\uncover<3->{%
   \item 
\KEY{Key Lemma} \ If a \CoR{} term $t$ distinguishes $d$ structures among $\M_1, \dots, \M_{2^k}$ from $\M_0$,
then $t$ needs $\ge d/2$ compositions.
\uncover<4->{%
\begin{center}
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{@{}w{l}{.6cm}|w{c}{.78cm}|*{7}{w{c}{.78cm}}|w{c}{.9cm}|w{c}{1.25cm}@{}}
  & $\M_0$ & $\M_{1}$ & $\M_{2}$ & $\M_{3}$ & $\M_{4}$ & $\M_{5}$ & $\cdots$ & $\M_{2^k}$ & \#diff. & \#comp. \\
  \hline
  $t$ & T & \alert{F} & T & \alert{F} & T & T & $\cdots$ & T & \alert{$2$} & $\ge 1$ \\
  $t'$ & F & \alert{T} & \alert{T} & \alert{T} & \alert{T} & F & $\cdots$ & F & \alert{$4$} & $\ge 2$ \\
\end{tabular}
\end{center}
}}
\end{itemize}

\vfill
\uncover<6->{\HP{if $t \equiv G_k$, then $t$ needs $\ge 2^k/2$ compositions.}}

\vfill
\uncover<7->{\large \KEY{TASK} How can we construct such $\M_0$ and $\M_1, \dots, \M_{2^k}$?}

\note{
\textbf{ページ概要}：
\begin{itemize}
  \item $G_k$ が $\M_0$ で成り立ち、$\M_1, \dots, \M_{2^k}$ で成り立たない構造を作る
  \item Key Lemma：$d$ 個を $\M_0$ から見分ける項は合成 $\ge d/2$ 個
\end{itemize}
}
\end{frame}

% The structure M_0 (edge-colored complete graph, two middle vertices, the hub)
\begin{frame}{Structures as colored complete graphs}

\KEY{View} structures as \KEY{complete graphs}, colored with $2^k$ colors $\mu \subseteq [k]$:
$P_i \iff i \in \mu$.
\\ \NOTE{$Q_i$ is used as the complement of $P_i$ (i.e., $i \notin \mu$): exactly one of $P_i, Q_i$ holds on each edge.}

\vfill
\begin{center}
\begin{tikzpicture}[every node/.style={font=\small}]
  % same style as the figure of G_k (example with k = 3): faint candidates for z with dashed edges
  % (the dashed edges are drawn only near the candidates, to leave room for the label of x-z)
  \node[pv,minimum size=7pt,label=left:$x$] (x) at (0,0) {};
  \node[pv,minimum size=7pt,label=right:$y$] (y) at (4.8,0) {};
  \foreach \h in {2.45,3.0} {
    \node[pv,minimum size=5pt,draw=black!30,fill=black!30] (c) at (3.0,\h) {};
    \draw[densely dashed,black!30,shorten <=2.2cm] (x) -- (c);
  }
  \node[font=\footnotesize,text=black!60] at (3.45,2.72) {$\exists z$};
  \node[pv,minimum size=7pt,label=right:$z$] (z) at (3.0,1.9) {};
  \draw[thick,cA] (x) -- node[below,align=center,text=cA]
    {color $\mu = \set{1, 3}$:\\ $P_1, \ Q_2, \ P_3$} (y);
  \draw[thick,cB] (x) -- node[pos=.45,above left,align=right,text=cB,font=\footnotesize]
    {color $\mu^c = \set{2}$:\\ $Q_1, \ P_2, \ Q_3$} (z);
  \node[anchor=west,align=left] at (5.8,1.0)
    {for every $i \in [3]$,\\
    the $x$-$y$ and $x$-$z$ edges\\
    have labels $P_i$ and $Q_i$, in either order};
\end{tikzpicture}
\end{center}

\vfill
\HP{$G_k(x, y)$ holds iff some $x$-$z$ edge has the color $\mu^c$, where $\mu$ is the color of the $x$-$y$ edge.}

\note{
\textbf{ページ概要}：
\begin{itemize}
  \item 構造を色付き完全グラフとして見る
  \item $G_k(x,y)$：$x$-$y$ 辺の色の補の色の $x$-$z$ 辺がある
\end{itemize}
}
\end{frame}

\begin{frame}{The randomly colored structure $\M_0$}

\begin{columns}[c]
  \begin{column}{.42\textwidth}
    \centering
    \input{figs/M0_random}

    \NOTE{$\M_0$: randomly colored, $O(k\,4^k)$ vertices, cf.~\fcite{jmt1995}}
  \end{column}
  \begin{column}{.55\textwidth}
    \KEY{Two-witness property} \ For all $u \ne v$ and all colors $\textcolor{cA}{\boldsymbol{\mu}}, \textcolor{cD}{\boldsymbol{\nu}}$,
    there are at least two \emph{witnesses} $w$ such that
    the $u$-$w$ and $w$-$v$ edges have colors $\textcolor{cA}{\boldsymbol{\mu}}$ and $\textcolor{cD}{\boldsymbol{\nu}}$, respectively.
  \end{column}
\end{columns}

\vfill
\pause
\vspace{.3em}
\KEY{Every \CoR{} term $t$ is color-invariant on $\M_0$}: $\M_0 \models t(u, v) \iff \M_0 \models t(u', v')$
whenever the $u$-$v$ and $u'$-$v'$ edges have the same color.

\vspace{1em}
Proof by induction on $t$: the case, $t = t_1 \mathbin{;} t_2$, follows from the (two-)witness property.

\note{
\textbf{ページ概要}：
\begin{itemize}
  \item $\M_0$：ランダム彩色、two-witness property
  \item だからどの項も color-invariant（帰納法、合成の場合に two-witness）
\end{itemize}
}
\end{frame}

\vfill
\onslide<4->{%
Flipping each of the $2^k$ colors $\lambda$ gives $\M_1, \dots, \M_{2^k}$.
% Each $\M_\lambda$ differs from $\M_0$ only at the hub, in one color.
}

\pause
\onslide<5->{%
$\leadsto$ Each composition newly distinguishes \alert{at most two} of them \NOTE{(next slides)}.
}
\note{
\textbf{ページ概要}：
\begin{itemize}
  \item hub の $\lambda$-edges を $\lambda^c$ に flip $\to$ hub に $\lambda$-edges がない
  \item $\M_0 \models G_k(\hub, v)$ だが $\M_\lambda \not\models G_k(\hub, v)$
  \item 色ごとに 1 つ、計 $2^k$ 個
\end{itemize}
}

\begin{frame}{Why a composition newly distinguishes at most two $\M_\lambda$ \ (1/2)}

Consider $t \mathbin{;} s$, where $t$ and $s$ do not distinguish $\M_\lambda$ from $\M_0$.\\
\KEY{Q.} For which pairs $(u, v)$ can \ $\M_0 \models (t \mathbin{;} s)(u, v) \iff \M_\lambda \models (t \mathbin{;} s)(u, v)$ \ fail?

$t$ are $s$ are color-invariant on both $\M_0$ and $\M_\lambda$:\\
$w$ is a witness depends only on the colors $(\mu, \nu)$ of the $u$-$w$ and $w$-$v$ edges.

\medskip
\KEY{Case $\hub \notin \{u, v\}$:} \ Never. \NOTE{(The equivalence is preserved.)}

% M_0 and M_lambda differ only in the edges at the hub (dashed on both sides);
% for every color pair, a witness can be re-picked away from the hub.
\begin{center}
\begin{tikzpicture}[every node/.style={font=\small},x=1.1cm,y=.68cm,
    lbl/.style={fill=normal text.bg,inner sep=2pt,font=\footnotesize},
    hubwit/.style={wit={$\hub$}}, hubplain/.style={nowit={$\hub$}},
    zwit/.style={witon=<3->{$w$}}, zplain/.style={nowit={$w$}}]
  \begin{scope}[visible on=<2->]
    % M_0: the witness is the hub; M_lambda: re-picked away from the hub, same color pair
    \foreach \dx/\str/\hst/\zst in {0/{\M_0}/hubwit/zplain, 4.2/{\M_\lambda}/hubplain/zwit} {
      \begin{scope}[xshift=\dx cm]
        \node at (1,1.95) {$\str$};
        \node[pairv] (u) at (0,0) {$u$};
        \node[pairv] (v) at (2,0) {$v$};
        \node (h) at (1,1.0) [\hst];
        \node (w) at (1,-1.0) [\zst];
        \draw[thick] (u) -- node[lbl] {$\mu$} (w);
        \draw[thick] (w) -- node[lbl] {$\nu$} (v);
      \end{scope}
    }
    \begin{pgfonlayer}{background}
      % (a layer does not inherit the scope's overlay, so repeat visible on)
      \begin{scope}[visible on=<2->]
      \draw[thick,black!35,densely dashed] (0,0) -- node[lbl,text=black] {$\mu$} (1,1.0);
      \draw[thick,black!35,densely dashed] (1,1.0) -- node[lbl,text=black] {$\nu$} (2,0);
      \draw[thick,black!35,densely dashed] (4.2/1.1,0) -- (4.2/1.1+1,1.0) -- (4.2/1.1+2,0);
      \end{scope}
    \end{pgfonlayer}
    \draw[-{Stealth},thick] (2.5,0) -- node[above,font=\scriptsize] {flip $\lambda$} (3.35,0);
    \node[anchor=west,font=\scriptsize,text=cW] at (1.22,1.2) {(picked as a witness)};
    % \node[anchor=west,align=left,font=\footnotesize,text=black!60] at (6.3,1.1) {dashed edges at $\hub$\\ are possibly changed by the flip};
  \end{scope}
  \node[font=\footnotesize,text=black!60,visible on=<3->, align= left] at (1,-1.85)
    {such $w$ exists by\\the \KEY{two-witness property}};
\end{tikzpicture}
\end{center}
\onslide<3->{
By the \KEY{two-witness property},\\
We can always pick a correspondence witness $w$ $(\ne \hub)$ with the same colors $(\mu, \nu)$.}

\note{
\textbf{ページ概要}：
\begin{itemize}
  \item $t$, $s$ がまだ見分けていないとき、$t \mathbin{;} s$ はどの点対で iff が崩れうるか
  \item $\M_0$ と $\M_\lambda$ の違いは hub の辺だけ
  \item $\hub \notin \{u, v\}$：どの色の組 $(\mu, \nu)$ でも hub 以外で witness を取り直せる（two witnesses）$\to$ iff は保たれる
\end{itemize}
}
\end{frame}

\begin{frame}{Why a composition newly distinguishes at most two $\M_\lambda$ \ (2/2)}

\KEY{Case $\hub \in \{u, v\}$,} say $u = \hub$: \ the edge $\hub$-$w$ itself may be flipped.
\NOTE{($v = \hub$: the same with $s$; \ $u = v = \hub$: similar)}

% Two rows: t holds only on lambda-edges (witness lost) / t also holds on a color mu != lambda
% (a mu-edge at the hub is not flipped). Colors as on the flipping slide: lambda cL, lambda^c cA.
\begin{center}
\begin{tikzpicture}[every node/.style={font=\small},x=1.1cm,y=.7cm,
    lbl/.style={fill=normal text.bg,inner sep=2pt,font=\footnotesize},
    zA/.style={witon=<2->{$w$}}, zB/.style={witon=<4->{$w$}}, zplain/.style={nowit={$w$}}]
  \foreach \row/\vis/\cv/\cz/\lz/\czl/\lzl/\zm/\zl/\tl/\cap in {
      0/1-/2-/cL/{\lambda}/cA/{\lambda^c}/zA/zplain/{t^{c}}/{if $t$ holds \emph{only} on $\lambda$-edges:\\ \alert{no witness} in $\M_\lambda$},
      -2.7/3-/4-/black/{\mu}/black/{\mu}/zB/zB/t/{if $t$ also holds on a color $\mu \ne \lambda$:\\ a $\mu$-edge at $\hub$ is not flipped}} {
    \begin{scope}[yshift=\row cm,visible on=<\vis>]
      \foreach \dx/\str/\col/\clab/\zst/\tlab in {0/{\M_0}/\cz/\lz/\zm/t, 3.6/{\M_\lambda}/\czl/\lzl/\zl/\tl} {
        \begin{scope}[xshift=\dx cm]
          \node at (1,2.1) {$\str$};
          \node[pairv] (h) at (0,0) {$\hub$};
          \node[pairv] (v) at (2,0) {$v$};
          \node (w) at (1,1.0) [\zst];
          \draw[thick,\col] (h) -- node[lbl,text=\col] {$\clab$} (w);
          \draw[thick] (w) -- node[lbl] {$\nu$} (v);
        \end{scope}
      }
      \draw[-{Stealth},thick] (2.2,.35) -- node[above,font=\scriptsize] {flip $\lambda$} (2.95,.35);
      \node[anchor=west,align=left,visible on=<\cv>] at (6.3,.35) {\cap};
    \end{scope}
  }
\end{tikzpicture}
\end{center}

\vfill
\onslide<5->{%
\begin{center}
  $\M_\lambda$ is newly distinguished only if $t$ or $s$ holds only on $\lambda$-edges
  \\ $\Rightarrow$ \ \alert{at most two} such $\lambda$ \ \NOTE{(one for $t$, one for $s$)}.
\end{center}}

\note{
\textbf{ページ概要}：
\begin{itemize}
  \item $\hub \in \{u, v\}$（例えば $u = \hub$）：$t$ が $\lambda$-edges だけで成り立つときだけ変わる（$s$ も同様）
  \item だから各合成が新たに見分ける $\M_\lambda$ は高々 2 個
  \item $(\hub, \hub)$ の場合も高々 2 は変わらない（論文参照）
\end{itemize}
}
\end{frame}
