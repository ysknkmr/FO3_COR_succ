\begin{frame}{The hard formula $G_k$}

\[
  G_k(x, y) \defeq \exists z \bigwedge_{i=1}^{k}
  \bigl[(P_i(x, y) \land Q_i(x, z)) \lor (Q_i(x, y) \land P_i(x, z))\bigr]
  \qquad \text{\NOTE{($|G_k| = O(k)$)}}
\]

\begin{center}
\begin{tikzpicture}[every node/.style={font=\small}]
  \node[pv,minimum size=7pt,label=left:$x$] (x) at (0,0) {};
  \node[pv,minimum size=7pt,label=right:$y$] (y) at (3.0,0) {};
  % z is existentially quantified: other candidate positions are drawn faintly,
  % stacked vertically so that the edges from x fan out without overlapping
  \foreach \h in {1.15,1.7} {
    \node[pv,minimum size=5pt,draw=black!30,fill=black!30] (c) at (2.0,\h) {};
    \draw[densely dashed,black!30] (x) -- (c);
  }
  \node[pv,minimum size=7pt,label=right:$z$] (z) at (2.0,.6) {};
  \draw[thick,cA] (x) -- (y);
  \draw[thick,cD] (x) -- (z);
  \node[font=\footnotesize,text=black!60] at (2.45,1.45) {$\exists z$};
\node[anchor=west,align=left] at (4.0,.6)
    {
    $G_k(x,y)$ holds iff there is $z$ such that\\
    for every $i \in [k]$,\\
    the $x$-$y$ and $x$-$z$ edges\\
    have labels $P_i$ and $Q_i$, in either order.};
\end{tikzpicture}
\end{center}

CoR: \ \EMPH{$2^k$ compositions} \quad \NOTE{(by DNF, as in the example before)}
\[
  G_k \equiv \bigcup_{A \subseteq [k]}
  \Bigl[ \Bigl(\bigcap_{i \in A} P_i \cap \bigcap_{i \notin A} Q_i\Bigr)
  \cap \Bigl(\Bigl(\bigcap_{i \in A} Q_i \cap \bigcap_{i \notin A} P_i\Bigr) \mathbin{;} \one\Bigr) \Bigr]
\]

\note{
\textbf{ページ概要}：
\begin{itemize}
  \item 下界を与える式 $G_k$ と、その DNF が合成 $2^k$ 個であること（2 倍を除いて最適）
\end{itemize}
}
