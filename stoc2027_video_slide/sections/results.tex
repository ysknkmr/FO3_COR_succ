\begin{frame}{The formula $G_k$}

\[
  G_k(x, y) \defeq \exists z \bigwedge_{i=1}^{k}
  \bigl[(P_i(x, y) \land Q_i(x, z)) \lor (Q_i(x, y) \land P_i(x, z))\bigr]
\]

\vfill
\begin{center}
\begin{tikzpicture}[every node/.style={font=\small}]
  \node[pv,minimum size=7pt,label=left:$x$] (x) at (0,0) {};
  \node[pv,minimum size=7pt,label=right:$y$] (y) at (3,0) {};
  \node[pv,minimum size=7pt,label=above:$z$] (z) at (1.3,1.6) {};
  \draw[thick,cA] (x) -- node[below] {read edge} (y);
  \draw[thick,cB] (x) -- node[above left] {witness edge} (z);
  \begin{scope}[visible on=<2->]
    \node[anchor=east,text=cA] at (6.2,1.1) {read edge};
    \node[anchor=east,text=cB] at (6.2,.3) {witness edge};
    \foreach \i/\r/\w in {1/P/Q,2/Q/P,3/P/Q,4/Q/P} {
      \node[text=cA] (r\i) at (6.2+.75*\i,1.1) {$\r_{\i}$};
      \node[text=cB] (w\i) at (6.2+.75*\i,.3) {$\w_{\i}$};
      \draw[black!30] (r\i) -- (w\i);
      \node[font=\scriptsize,text=black!50] at (6.2+.75*\i,1.6) {$i = \i$};
    }
  \end{scope}
\end{tikzpicture}
\end{center}

\vfill
\onslide<2->{%
\begin{center}
  at every coordinate $i$: one edge meets $P_i$, the other meets $Q_i$
  \ $\Rightarrow$ \ the read edge \emph{accepts} the witness edge
\end{center}}

\note{
\textbf{伝えること}：下界を与える式 $G_k$ の読み方。read edge が witness edge を「受理する」$=$ 各座標 $i$ で一方が $P_i$、他方が $Q_i$。
\begin{itemize}
  \item 量化子 1 つ、サイズ $8k$ の小さな式
  \item 右の表で「座標ごとに $P$ と $Q$ が入れ替わる」ことを見せる
\end{itemize}
}
\end{frame}

    node[midway,below=8pt] {$G_k \equiv \bigcup_{A \subseteq [k]} \bigl[ H_A \cap (H_{A^c} \mathbin{;} \one) \bigr]$ \ : \ \EMPH{$2^k$ compositions}};
\end{tikzpicture}
\end{center}
\NOTE{\hfill $H_A \defeq \bigcap_{i \in A} P_i \cap \bigcap_{i \notin A} Q_i$; \
a read edge in $H_A$ accepts every witness edge in $H_{A^c}$}

\vfill

\note{
\textbf{伝えること}：座標ごとの順序を固定すると、$2^k$ 個の合成を使う DNF で書ける。問い：これより本質的に少なくできるか？
\begin{itemize}
  \item $H_A$ の read edge は $H_{A^c}$ の witness edge をすべて受理する
  \item $X \mathbin{;} \one$ は「$x$ から $X$ の辺が出ている」
\end{itemize}
}
\end{frame}

\begin{frame}{Main results}

\begin{theorem}[Exponential lower bounds for full \CoR]
  Every relational circuit $C \eqfin G_k$ has $\ge 2^{k-1}$ composition gates.
\end{theorem}

\vfill
\pause
\begin{center}
\begin{tikzpicture}[every node/.style={font=\small}]
  \draw[thick,-{Stealth}] (0,0) -- (10,0) node[right] {\#compositions};
  \foreach \x/\l in {2/{$O(k)$},6/{$2^{k-1}$},8.5/{$2^k$}} {
    \draw (\x,.12) -- (\x,-.12); \node[below=3pt] at (\x,0) {\l};
  }
  \node[above=4pt] at (2,0) {$|G_k|$};
  \node[above=4pt,text=chibasub] at (6,0) {lower bound};
  \node[above=4pt] at (8.5,0) {DNF};
\end{tikzpicture}
\end{center}

\vfill
\pause
\begin{itemize}
  \item shared subterms, complement, converse: all allowed and free
  \item also: term size $\Theta(k 2^k)$; \ the blowup begins at domain size $2^{\Theta(k)}$
\end{itemize}

\note{
\textbf{伝えること}：主定理。$G_k$ と同値な関係回路は $2^{k-1}$ 個以上の合成を使う。
\begin{itemize}
  \item 共有・補・逆はすべて許して\emph{無料}。数えるのは合成（行列積）だけ
  \item サイズ $O(k)$ の式から $2^{\Omega(k)}$ 個の合成が必要
  \item 補足（さらっと）：項サイズは $\Theta(k 2^k)$、爆発は領域サイズ $2^{\Theta(k)}$ から始まる
\end{itemize}
}
\end{frame}
