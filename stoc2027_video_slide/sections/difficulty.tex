\section{Natural approaches that fail}

% ============================================================
\begin{frame}{$\Sigma_1$\CoR{}: organized as a fooling-set argument}

Before this work, exponential lower bounds were known only for $\Sigma_1$\CoR, even with shared subterms
\NOTE{(positive \CoR{} \cite{nakamura2020}; $\Sigma_1$\CoR{} \cite[Theorems 6.1 and 6.12]{nakamura2022})}.\\
\NOTE{$\Sigma_1$\CoR: complement may be applied only to input relations and $\id$, and so never stands above a composition.}

\vspace{.3em}
\begin{columns}[T]
\begin{column}{.47\textwidth}
\begin{block}{Communication complexity}
  $F \subseteq f^{-1}(1)$ is a \emph{fooling set} if any two
  $(a, b), (a', b') \in F$ have a crossed input with value $0$.\\
  Covering $F$ by 1-rectangles takes $\ge |F|$ of them.
\end{block}
\end{column}
\begin{column}{.50\textwidth}
\pause
\begin{block}{Nakamura's instances}
  $t = \Theta(2^k)$ structures $\mathcal S_i$ on $u_i, w_i, v_i$; \ one $\mathcal N$ with $u^*, v^*$:
  \begin{itemize}
    \item $\sem{\varphi}{\mathcal S_i}(u_i, v_i) = 1$, \ $\sem{\varphi}{\mathcal N}(u^*, v^*) = 0$;
    \item for $i < j$, some $c$ with $(\mathcal S_i; u_i, w_i) \hookrightarrow (\mathcal N; u^*, c)$
      and $(\mathcal S_j; w_j, v_j) \hookrightarrow (\mathcal N; c, v^*)$.
  \end{itemize}
\end{block}
\end{column}
\end{columns}

\vfill
\pause
\begin{itemize}
  \item $\Sigma_1$\CoR{} is \emph{preserved under extensions}: \
    $\iota(\sem{T}{\mathcal S}) \subseteq \sem{T}{\mathcal S'}$ for every embedding $\iota$. \hfill $(\diamond)$
  \item A lowest subterm separating $\mathcal S_i$ from $\mathcal N$ is a composition $X \mathbin{;} Y$
    with witness $w_i$: a 1-rectangle.
  \item If $\mathcal S_i, \mathcal S_j$ share it, $c$ is a witness for $(u^*, v^*)$ in $\mathcal N$ by $(\diamond)$.
    \ $\Rightarrow$ \ $t$ compositions.
\end{itemize}

\note{
- Before this work, exponential lower bounds were known only for Sigma_1 CoR.
- Organized as a fooling-set argument, the proof is simple.
- Nakamura's instances: many small structures that satisfy the query, and one large structure that does not,
  into which crossed pairs embed.
- Sigma_1 CoR is preserved under extensions, so each separating composition acts as a 1-rectangle,
  and no two instances share one.
}
\end{frame}

% ============================================================
\begin{frame}{With complement, both ingredients fail}

\begin{enumerate}
  \item \emph{Complement reverses $(\diamond)$}:
    \ $\overline{R \mathbin{;} \one}$ holds at $(u, v)$ when $u$ has no $R$-successor,
    but fails at $(\iota(u), \iota(v))$ once $\iota(u)$ gains one.
  \vfill
  \pause
  \item \emph{A term with two compositions separates every $\mathcal S_i$ from $\mathcal N$}:
    \[
      \overline{(E \cap (E \mathbin{;} \overline{\id}))} \mathbin{;} \one
      \qquad \text{\NOTE{($E$: the adjacency relation)}}
    \]
\end{enumerate}

\vfill
\pause
\NOTE{Other lower-bound methods do not apply directly either:
formula-size and EF/pebble games ($G_k$ needs one quantifier and three variables; its DNF nests no compositions);
rank and discrepancy (free Boolean operations defeat measures of a single matrix).}

\vfill
\begin{alertblock}{}
  \centering
  Unlike the inclusion $(\diamond)$, an \emph{equality} survives complement.
\end{alertblock}

\note{
- Allowing complement above compositions breaks this argument completely.
- First, complement reverses the inclusion (diamond).
- Second, a term with two compositions separates every S_i from N.
- Other methods do not apply directly either.
- Unlike the inclusion (diamond), an equality survives complement.
}
\end{frame}
