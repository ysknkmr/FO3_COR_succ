\begin{fact}[for $\Sigma_1$\CoR]
  If $\M$ is an induced substructure of $\M'$, then $T^{\M} \subseteq T^{\M'}$.
\end{fact}
\NOTE{A witness in $\M$ remains a witness in $\M'$.
\fcite{nakamura2022} combines this with rectangularity of $;$ (a fooling-set argument).}

\vfill
\pause
\begin{alertblock}{Complement reverses inclusion}
  $(u, u) \in \overline{R \mathbin{;} \one}$ \ iff \ $u$ has no outgoing $R$-edge:
  \ true in $\M$, \ false in $\M'$.
\end{alertblock}

\note{
- Previous lower bounds compare a small structure with a larger structure that contains it.
- Without complement above composition, everything true in the small structure stays true in the larger one.
- With complement this fails: "u has no outgoing R-edge" is destroyed by adding an edge.
- On the instances of [Nakamura 2022], one term with complement separates all of them at once.
- Any argument that transfers tests into a larger structure breaks down for full CoR.
}
\end{frame}

\begin{frame}{Symmetric structures are infinite}

\begin{columns}[T]
\begin{column}{.48\textwidth}
\begin{block}{Infinite homogeneous colored graph}
  Every \CoR-definable relation is invariant under automorphisms,
  hence a union of color classes (and possibly $\id$).\\[.2em]
  Complement is harmless.
\end{block}
\end{column}
\begin{column}{.48\textwidth}
\pause
\begin{alertblock}{Correctness on finite structures}
  An infinite counter-model does not exclude a short term that is correct
  on all \emph{finite} structures.\\[.2em]
  Finite homogeneous structures are rare.
\end{alertblock}
\end{column}
\end{columns}

\vfill
\pause
\begin{center}
  We need a comparison that is preserved by complement,\\
  carried out on a \emph{finite} structure.
\end{center}

\note{
- In a homogeneous (Fraisse limit) colored graph, automorphisms map any edge to any other edge of the same color,
  so every intermediate relation is a union of colors, even with complement.
- But a lower bound for finite structures needs finite counter-models, and finite homogeneous structures are rare.
}
\end{frame}
